33/// `TimeoutLayer` adds deadlines to operations so slow or stalled work cannot
34/// hang indefinitely.
35///
36/// For example, a dead connection could stall a database query. `TimeoutLayer`
37/// interrupts the operation and returns an error that users can retry or report.
38///
39/// # Notes
40///
41/// `TimeoutLayer` has two default timeouts.
42///
43/// - `timeout` bounds asynchronous service calls such as `stat`, `create_dir`,
44///   `rename`, `restore`, and `presign`.
45/// - `io_timeout` bounds each I/O body method and each concurrent I/O task that
46///   uses the executor's timeout. This includes opening a read range, reading
47///   from its stream, fetching a list entry, and uploading a multipart part.
48///
49/// [`with_operation_timeout`](Self::with_operation_timeout) replaces the
50/// applicable default for the specified operation.
51///
52/// # Default
53///
54/// - timeout: 60 seconds
55/// - io_timeout: 10 seconds
56///
57/// # Cancellation Safety
58///
59/// `TimeoutLayer` enforces deadlines by dropping the in-flight future when a
60/// timeout is reached. This can break lower layers that rely on a future being
61/// resolved to restore internal state.
62///
63/// For example, while using `TimeoutLayer` with `RetryLayer` at the same time,
64/// please make sure timeout layer is added before retry layer.
65///
66/// ```no_run
67/// # use std::time::Duration;
68/// #
69/// # use opendal_core::services;
70/// # use opendal_core::Operator;
71/// # use opendal_core::Result;
72/// # use opendal_layer_retry::RetryLayer;
73/// # use opendal_layer_timeout::TimeoutLayer;
74/// #
75/// # fn main() -> Result<()> {
76/// let op = Operator::new(services::Memory::default())?
77///     // This is fine: each retry attempt is timed out.
78///     .layer(TimeoutLayer::default().with_io_timeout(Duration::from_nanos(1)))
79///     .layer(RetryLayer::default())
80///     // This is wrong: timeout can drop RetryLayer's future before it restores body state.
81///     .layer(TimeoutLayer::default().with_io_timeout(Duration::from_nanos(1)));
82/// # Ok(())
83/// # }
84/// ```
85///
86/// # Examples
87///
88/// The following example sets defaults of 10 seconds for control operations
89/// and 3 seconds for I/O. Each lister call may take up to 15 seconds.
90///
91/// ```no_run
92/// # use std::time::Duration;
93/// #
94/// use opendal_core::raw::Operation;
95/// # use opendal_core::services;
96/// # use opendal_core::Operator;
97/// # use opendal_core::Result;
98/// # use opendal_layer_timeout::TimeoutLayer;
99/// #
100/// # fn main() -> Result<()> {
101/// let _ = Operator::new(services::Memory::default())?
102///     .layer(
103///         TimeoutLayer::default()
104///             .with_timeout(Duration::from_secs(10))
105///             .with_io_timeout(Duration::from_secs(3))
106///             .with_operation_timeout(Operation::List, Duration::from_secs(15)),
107///     );
108/// # Ok(())
109/// # }
110/// ```
111///
112/// # Implementation Notes
113///
114/// `TimeoutLayer` uses the configured executor's timer to bound service calls,
115/// I/O body methods, and individual concurrent I/O tasks. It uses Tokio timers
116/// when the executor does not implement [`Execute::sleep`].
117///
118/// This introduces a small amount of overhead for IO operations, but it is needed
119/// to implement timeouts correctly. OpenDAL used to implement this as a
120/// zero-cost deadline check that only stored an [`Instant`] and compared it with
121/// the current time. However, that approach does not work for all cases.
122///
123/// For example, a user's TCP connection could enter the
124/// [Busy ESTAB](https://blog.cloudflare.com/when-tcp-sockets-refuse-to-die)
125/// state. In this state, the connection emits no IO events, so the runtime never
126/// polls the future again. The future hangs until Linux closes the connection
127/// after it reaches the
128/// [net.ipv4.tcp_retries2](https://man7.org/linux/man-pages/man7/tcp.7.html)
129/// limit.
130#[derive(Clone, Debug)]
131pub struct TimeoutLayer {
132    timeout: Duration,
133    io_timeout: Duration,
134    operation_timeouts: HashMap<Operation, Duration>,
135}
136
137impl Default for TimeoutLayer {
138    fn default() -> Self {
139        Self {
140            timeout: Duration::from_secs(60),
141            io_timeout: Duration::from_secs(10),
142            operation_timeouts: HashMap::new(),
143        }
144    }
145}
146
147impl TimeoutLayer {
148    /// Create a new [`TimeoutLayer`] with default settings.
149    pub fn new() -> Self {
150        Self::default()
151    }
152
153    /// Set the default timeout for control operations such as `stat` and `rename`.
154    pub fn with_timeout(mut self, timeout: Duration) -> Self {
155        self.timeout = timeout;
156        self
157    }
158
159    /// Set the default timeout for I/O body methods and concurrent I/O tasks.
160    pub fn with_io_timeout(mut self, timeout: Duration) -> Self {
161        self.io_timeout = timeout;
162        self
163    }
164
165    /// Set the timeout for the specified operation.
166    ///
167    /// Control operations use this value in place of [`with_timeout`](Self::with_timeout).
168    /// Operations with I/O bodies such as read, write, list, delete, copy, and compose
169    /// use it in place of [`with_io_timeout`](Self::with_io_timeout) for each async
170    /// body method and each concurrent I/O task that uses the executor's timeout.
171    /// For example, a write override applies to each write, close, or abort call
172    /// and each multipart upload task separately. Synchronous calls such as
173    /// [`Operation::Info`] and I/O body construction have no deadline.
174    ///
175    /// Each call replaces any previous override for the same operation.
176    pub fn with_operation_timeout(mut self, operation: Operation, timeout: Duration) -> Self {
177        self.operation_timeouts.insert(operation, timeout);
178        self
179    }
180}

198    inner: Servicer,
199    config: TimeoutLayer,
200}

202impl TimeoutService {
203    async fn timeout<F: Future<Output = Result<T>>, T>(
204        &self,
205        ctx: &OperationContext,
206        op: Operation,
207        fut: F,
208    ) -> Result<T> {
209        let timeout = self
210            .config
211            .operation_timeouts
212            .get(&op)
213            .copied()
214            .unwrap_or(self.config.timeout);
215        execute_with_timeout(
216            ctx.executor().inner(),
217            timeout,
218            op.into_static(),
219            "operation timeout reached",
220            fut,
221        )
222        .await
223    }
224
225    fn io_context<'a>(
226        &self,
227        ctx: &'a OperationContext,
228        op: Operation,
229    ) -> (Duration, Cow<'a, OperationContext>) {
230        let Some(&timeout) = self.config.operation_timeouts.get(&op) else {
231            return (self.config.io_timeout, Cow::Borrowed(ctx));
232        };
233
234        // Bodies such as MultipartWriter retain the executor for their individual tasks.
235        (
236            timeout,
237            Cow::Owned(TimeoutExecutor::wrap_context(ctx, timeout)),
238        )
239    }
240}

366impl TimeoutExecutor {
367    fn wrap_context(ctx: &OperationContext, timeout: Duration) -> OperationContext {
368        ctx.with_executor(Executor::with(Self {
369            exec: ctx.executor().clone().into_inner(),
370            timeout,
371        }))
372    }
373}

396    timeout: Duration,

400impl<R> TimeoutWrapper<R> {
401    fn new(inner: R, timeout: Duration, executor: Executor) -> Self {
402        Self {
403            inner,
404            timeout,
405            executor,
406        }
407    }
408
409    async fn io_timeout<F: Future<Output = Result<T>>, T>(
410        executor: &Executor,
411        timeout: Duration,
412        op: Operation,
413        fut: F,
414    ) -> Result<T> {
415        execute_with_timeout(
416            executor.inner(),
417            timeout,
418            op.into_static(),
419            "io operation timeout reached",
420            fut,
421        )
422        .await
423    }
424}
425
426fn timer(executor: &dyn Execute, duration: Duration) -> BoxedStaticFuture<()> {
427    executor
428        .sleep(duration)
429        .unwrap_or_else(|| Box::pin(tokio::time::sleep(duration)))
430}
431
432async fn execute_with_timeout<F: Future<Output = Result<T>>, T>(
433    executor: &dyn Execute,
434    duration: Duration,
435    operation: &'static str,
436    message: &'static str,
437    fut: F,
438) -> Result<T> {
439    let deadline = timer(executor, duration);
440    futures::pin_mut!(fut);
441    match select(fut, deadline).await {
442        Either::Left((result, _)) => result,
443        Either::Right(_) => Err(Error::new(ErrorKind::Unexpected, message)
444            .with_operation(operation)
445            .with_context("timeout", duration.as_secs_f64().to_string())
446            .set_temporary()),
447    }
448}

824    async fn test_read_and_concurrent_multipart_write_timeout_overrides() {
825        for (part_secs, should_succeed) in [(3, true), (6, false)] {
826            let service = MockService {
827                write_part_delay: Duration::from_secs(part_secs),
828                ..Default::default()
829            };
830            let ctx = OperationContext::default().with_executor(Executor::with(TestExecutor));
831            let op = Operator::from_parts(ctx, Arc::new(service)).layer(
832                TimeoutLayer::new()
833                    .with_io_timeout(Duration::from_secs(2))
834                    .with_operation_timeout(Operation::Read, Duration::from_secs(1))
835                    .with_operation_timeout(Operation::Write, Duration::from_secs(5)),
836            );
837
838            let reader = op.reader("test").await.unwrap();
839            let start = Instant::now();
840            let err = reader.read(0..4).await.unwrap_err();
841            assert_eq!(err.kind(), ErrorKind::Unexpected);
842            assert_eq!(err.message(), "io operation timeout reached");
843            assert_eq!(start.elapsed().as_secs(), 1);
844
845            let mut writer = op.writer_with("test").concurrent(2).await.unwrap();
846            writer.write("a").await.unwrap();
847            let start = Instant::now();
848            writer.write("b").await.unwrap();
849            // Start closing after the first part's timer has started so the
850            // part deadline can expire before the close deadline.
851            sleep(Duration::from_secs(1)).await;
852            let result = writer.close().await;
853            if should_succeed {
854                assert_eq!(result.unwrap().content_length(), 2);
855            } else {
856                let err = result.unwrap_err();
857                assert_eq!(err.kind(), ErrorKind::Unexpected);
858                assert_eq!(err.message(), "write part timeout");
859                assert_eq!(start.elapsed().as_secs(), 5);
860            }
861        }
862    }
