48impl Debug for SftpCore {
49    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
50        f.debug_struct("SftpCore")
51            .field("endpoint", &self.endpoint)
52            .field("root", &self.root)
53            .finish_non_exhaustive()
54    }
55}

57impl SftpCore {
58    pub fn new(
59        info: ServiceInfo,
60        capability: Capability,
61        manager: Manager,
62        acquire_timeout: Option<Duration>,
63    ) -> Self {
64        Self {
65            info,
66            capability,
67            endpoint: manager.endpoint.clone(),
68            root: manager.root.clone(),
69            client: unbounded::Pool::new(unbounded::PoolConfig::default(), manager),
70            capacity: Arc::new(Semaphore::new(POOL_CAPACITY)),
71            acquire_timeout,
72        }
73    }
74
75    pub async fn connect(&self) -> Result<SftpConnection> {
76        let permit = connection_stage(self.acquire_timeout, "waiting for pool capacity", async {
77            Ok(self.capacity.clone().acquire_owned(1).await)
78        })
79        .await?;
80        let object = self.client.get().await?;
81        Ok(SftpConnection {
82            object,
83            _permit: permit,
84        })
85    }
86}

102pub struct Manager {
103    pub(super) endpoint: String,
104    pub(super) root: String,
105    pub(super) user: Option<String>,
106    pub(super) key: Option<String>,
107    pub(super) known_hosts_strategy: KnownHosts,
108    pub(super) connect_timeout: Option<Duration>,
109    #[cfg(not(windows))]
110    pub(super) control_directory: Option<String>,
111}
112
113impl Manager {
114    async fn initialize_root(&self, sftp: &Sftp) -> Result<()> {
115        let mut fs = sftp.fs();
116        let mut current = PathBuf::new();
117        for component in Path::new(&self.root).components() {
118            let path = component.as_os_str();
119            current.push(path);
120            if component != Component::RootDir
121                && let Err(error) = fs.create_dir(path).await
122            {
123                // Servers may reject mkdir even when an existing directory can be used.
124                if !matches!(
125                    error,
126                    SftpClientError::SftpError(
127                        SftpErrorKind::Failure | SftpErrorKind::PermDenied,
128                        _
129                    )
130                ) {
131                    return Err(parse_sftp_error(error));
132                }
133                match fs.metadata(path).await {
134                    Ok(metadata) if metadata.file_type().is_some_and(|kind| kind.is_dir()) => {}
135                    Err(err) if classify_sftp_error(&err).1 => return Err(parse_sftp_error(err)),
136                    _ => return Err(parse_sftp_error(error)),
137                }
138            }
139            fs.set_cwd(&current);
140        }
141        Ok(())
142    }
143}

145impl ManageObject for Manager {
146    type Object = Sftp;
147    type Error = Error;
148
149    async fn create(&self) -> Result<Self::Object, Self::Error> {
150        let mut session = SessionBuilder::default();
151
152        if let Some(user) = &self.user {
153            session.user(user.clone());
154        }
155
156        if let Some(key) = &self.key {
157            session.keyfile(key);
158        }
159
160        session.known_hosts_check(self.known_hosts_strategy.clone());
161        #[cfg(not(windows))]
162        if let Some(dir) = &self.control_directory {
163            session.control_directory(dir);
164        }
165        if let Some(timeout) = self.connect_timeout {
166            // OpenSSH uses whole seconds. Keep a process-side limit even if the Rust
167            // deadline expires first and drops the connect future.
168            let seconds = timeout
169                .as_secs()
170                .saturating_add(u64::from(timeout.subsec_nanos() != 0))
171                .min(i32::MAX as u64);
172            session.connect_timeout(Duration::from_secs(seconds));
173        }
174
175        let session = connection_stage(
176            self.connect_timeout,
177            "SSH connection and authentication",
178            async {
179                session
180                    .connect(&self.endpoint)
181                    .await
182                    .map_err(parse_ssh_error)
183            },
184        )
185        .await?;
186
187        let sftp = connection_stage(
188            self.connect_timeout,
189            "SFTP subsystem initialization",
190            async {
191                Sftp::from_session(session, SftpOptions::default())
192                    .await
193                    .map_err(parse_sftp_error)
194            },
195        )
196        .await?;
197
198        connection_stage(
199            self.connect_timeout,
200            "SFTP root initialization",
201            self.initialize_root(&sftp),
202        )
203        .await
204        .map_err(|err| err.with_context("root", &self.root))?;
205
206        debug!("sftp connection created at {}", self.root);
207        Ok(sftp)
208    }
209
210    async fn is_recyclable(
211        &self,
212        o: &mut Self::Object,
213        _: &ObjectStatus,
214    ) -> Result<(), Self::Error> {
215        let mut fs = o.fs();
216        fs.set_cwd(&self.root);
217        let result = connection_stage(
218            self.connect_timeout,
219            "SFTP connection health check",
220            async {
221                fs.metadata("./").await.map_err(parse_sftp_error)?;
222                Ok(())
223            },
224        )
225        .await;
226        if let Err(err) = &result {
227            // The pool discards the failed connection and tries another checkout.
228            debug!("discarding pooled SFTP connection: {err:?}");
229        }
230        result
231    }
232}
233
234async fn connection_stage<T>(
235    timeout: Option<Duration>,
236    stage: &'static str,
237    future: impl Future<Output = Result<T>>,
238) -> Result<T> {
239    let result = match timeout {
240        Some(timeout) => tokio::time::timeout(timeout, future)
241            .await
242            .unwrap_or_else(|err| {
243                Err(
244                    Error::new(ErrorKind::Unexpected, "SFTP connection timed out")
245                        .set_temporary()
246                        .set_source(err),
247                )
248            }),
249        None => future.await,
250    };
251    result.map_err(|err| err.with_context("stage", stage))
252}

254pub fn parse_sftp_error(e: SftpClientError) -> Error {
255    let (kind, temporary) = classify_sftp_error(&e);
256    let error = Error::new(kind, "sftp error").set_source(e);
257    if temporary {
258        error.set_temporary()
259    } else {
260        error
261    }
262}
263
264fn classify_sftp_error(error: &SftpClientError) -> (ErrorKind, bool) {
265    match error {
266        SftpClientError::UnsupportedSftpProtocol { .. }
267        | SftpClientError::SftpError(SftpErrorKind::OpUnsupported, _) => {
268            (ErrorKind::Unsupported, false)
269        }
270        SftpClientError::SftpError(SftpErrorKind::NoSuchFile, _) => (ErrorKind::NotFound, false),
271        SftpClientError::SftpError(SftpErrorKind::PermDenied, _) => {
272            (ErrorKind::PermissionDenied, false)
273        }
274        SftpClientError::SftpError(SftpErrorKind::Failure, _) => (ErrorKind::Unexpected, true),
275        SftpClientError::IOError(err) => classify_io_error(err),
276        SftpClientError::BackgroundTaskFailure(_) => (ErrorKind::Unexpected, true),
277        SftpClientError::RemoteChildSpawnError(err) => classify_ssh_error(err),
278        SftpClientError::RecursiveErrors(errors) => classify_sftp_error(&errors.original_error),
279        SftpClientError::RecursiveErrors3(errors) => classify_sftp_error(&errors.err1),
280        _ => (ErrorKind::Unexpected, false),
281    }
282}

284pub fn parse_ssh_error(e: SshError) -> Error {
285    let (kind, temporary) = classify_ssh_error(&e);
286    let error = Error::new(kind, "failed to establish SSH connection").set_source(e);
287    if temporary {
288        error.set_temporary()
289    } else {
290        error
291    }
292}
293
294fn classify_ssh_error(error: &SshError) -> (ErrorKind, bool) {
295    match error {
296        SshError::Disconnected | SshError::RemoteProcessTerminated => (ErrorKind::Unexpected, true),
297        SshError::Connect(err)
298            if matches!(
299                err.kind(),
300                io::ErrorKind::ConnectionAborted | io::ErrorKind::Other
301            ) =>
302        {
303            // openssh uses ConnectionAborted for unrecognized diagnostics, including
304            // host key and authentication failures as well as network failures.
305            let diagnostic = err.to_string();
306            if diagnostic.lines().any(|line| {
307                line == "Host key verification failed." || line.contains(": Permission denied (")
308            }) {
309                return (ErrorKind::PermissionDenied, false);
310            }
311            let temporary = diagnostic.lines().any(|raw_line| {
312                let line = raw_line
313                    .strip_prefix("kex_exchange_identification: ")
314                    .or_else(|| raw_line.strip_prefix("ssh_exchange_identification: "))
315                    .unwrap_or(raw_line);
316                let line = line.strip_prefix("read: ").unwrap_or(line);
317                matches!(
318                    line,
319                    "Connection timed out during banner exchange"
320                        | "Connection timed out"
321                        | "Operation timed out"
322                        | "Broken pipe"
323                ) || raw_line.ends_with(" Connection timed out")
324                    || raw_line.ends_with(" Operation timed out")
325                    || raw_line.ends_with(": Network is unreachable")
326                    || raw_line.ends_with(": No route to host")
327                    || raw_line.ends_with(" Connection reset by peer")
328                    || raw_line.ends_with(" Broken pipe")
329                    || line.starts_with("Connection closed by ")
330                    || line.starts_with("Connection reset by ")
331            });
332            (ErrorKind::Unexpected, temporary)
333        }
334        SshError::Connect(err)
335        | SshError::Master(err)
336        | SshError::Ssh(err)
337        | SshError::Remote(err)
338        | SshError::ChildIo(err) => classify_io_error(err),
339        _ => (ErrorKind::Unexpected, false),
340    }
341}
342
343fn classify_io_error(error: &io::Error) -> (ErrorKind, bool) {
344    match error.kind() {
345        io::ErrorKind::PermissionDenied => (ErrorKind::PermissionDenied, false),
346        io::ErrorKind::NotFound => (ErrorKind::NotFound, false),
347        io::ErrorKind::TimedOut
348        | io::ErrorKind::ConnectionRefused
349        | io::ErrorKind::ConnectionReset
350        | io::ErrorKind::ConnectionAborted
351        | io::ErrorKind::NotConnected
352        | io::ErrorKind::NetworkUnreachable
353        | io::ErrorKind::HostUnreachable
354        | io::ErrorKind::BrokenPipe
355        | io::ErrorKind::UnexpectedEof => (ErrorKind::Unexpected, true),
356        _ => (ErrorKind::Unexpected, false),
357    }
358}

372mod utils {
373    use openssh_sftp_client::metadata::MetaData as SftpMeta;
374
375    use opendal_core::EntryMode;
376    use opendal_core::Error;
377    use opendal_core::ErrorKind;
378    use opendal_core::Metadata;
379    use opendal_core::MetadataBuilder;
380    use opendal_core::Result;
381    use opendal_core::raw::Timestamp;
382
383    pub fn to_metadata(meta: SftpMeta) -> Result<Metadata> {
384        let mode = meta
385            .file_type()
386            .map(|filetype| {
387                if filetype.is_file() {
388                    EntryMode::FILE
389                } else if filetype.is_dir() {
390                    EntryMode::DIR
391                } else {
392                    EntryMode::Unknown
393                }
394            })
395            .unwrap_or(EntryMode::Unknown);
396
397        let mut metadata = match mode {
398            EntryMode::FILE => MetadataBuilder::file(meta.len().ok_or_else(|| {
399                Error::new(
400                    ErrorKind::Unexpected,
401                    "sftp metadata does not contain file length",
402                )
403            })?),
404            EntryMode::DIR => MetadataBuilder::dir(),
405            EntryMode::Unknown => MetadataBuilder::unknown(),
406        };
407
408        if let Some(modified) = meta.modified()
409            && let Ok(m) = Timestamp::try_from(modified.as_system_time())
410        {
411            metadata.last_modified(m);
412        }
413
414        Ok(metadata.build())
415    }

453    }
