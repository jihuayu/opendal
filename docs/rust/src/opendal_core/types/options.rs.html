25/// Options for delete operations.
26///
27/// Each condition checks the file currently stored at the delete path, and
28/// every condition must hold for the delete to proceed:
29///
30/// - A false condition fails the delete with
31///   [`crate::ErrorKind::ConditionNotMatch`].
32/// - A condition the service does not advertise through its capability fails
33///   the delete with [`crate::ErrorKind::Unsupported`]; OpenDAL never
34///   silently drops a condition. A service that cannot preserve a
35///   combination of conditions also returns `Unsupported`.
36/// - A service-side conflict unrelated to these conditions surfaces as
37///   [`crate::ErrorKind::Conflict`].
38///
39/// Once every condition holds, deleting a missing file remains a successful
40/// no-op.
41///
42/// See the [conditional operation specification][crate::docs::specs::conditional_operations]
43/// for the complete cross-operation contract.
44#[derive(Debug, Clone, Default, Eq, PartialEq)]
45pub struct DeleteOptions {
46    /// Delete the given version of the file instead of the current one.
47    ///
48    /// This selects which stored version the delete removes; it is not a
49    /// condition on the file.
50    ///
51    /// Check [`crate::Capability::delete_with_version`] before using this
52    /// option.
53    pub version: Option<String>,
54    /// Whether to delete the target recursively.
55    ///
56    /// The default, `false`, deletes a single object. With `true`, a trailing
57    /// slash (for example, `"a/"`) scopes deletion to a directory and its
58    /// descendants across services. Prefix matching for paths without a
59    /// trailing slash is service-specific. On prefix-based services such as
60    /// `memory`, `"a"` also matches entries under `"ab/"`.
61    pub recursive: bool,
62    /// Delete only when the file at the delete path has this exact ETag.
63    ///
64    /// The condition succeeds when the file exists and its ETag equals this
65    /// value. A file with a different ETag, or a missing file, fails the
66    /// delete with [`crate::ErrorKind::ConditionNotMatch`]. Only concrete
67    /// ETag values are portable; a wildcard such as `"*"` has no portable
68    /// meaning here.
69    ///
70    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
71    /// service does not advertise [`crate::Capability::delete_with_if_match`].
72    pub if_match: Option<String>,
73    /// Delete only when the file at the delete path does not have this ETag.
74    ///
75    /// With a concrete ETag value, the condition succeeds when the file has
76    /// a different ETag or does not exist; deleting a missing file is then a
77    /// successful no-op. A file whose ETag equals this value fails the
78    /// delete with [`crate::ErrorKind::ConditionNotMatch`]. Only concrete
79    /// ETag values are portable; a wildcard such as `"*"` has no portable
80    /// meaning here.
81    ///
82    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
83    /// service does not advertise
84    /// [`crate::Capability::delete_with_if_none_match`].
85    pub if_none_match: Option<String>,
86    /// Delete only when the file at the delete path has this exact version.
87    ///
88    /// The condition succeeds when the file exists and its version equals
89    /// this value. A file with a different version, or a missing file, fails
90    /// the delete with [`crate::ErrorKind::ConditionNotMatch`].
91    ///
92    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
93    /// service does not advertise
94    /// [`crate::Capability::delete_with_if_version_match`].
95    pub if_version_match: Option<String>,
96    /// Delete only when the file at the delete path does not have this
97    /// version.
98    ///
99    /// The condition succeeds when the file exists with a different version.
100    /// A file whose version equals this value fails the delete with
101    /// [`crate::ErrorKind::ConditionNotMatch`]. When no file exists at the
102    /// delete path, no portable behavior is defined.
103    ///
104    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
105    /// service does not advertise
106    /// [`crate::Capability::delete_with_if_version_not_match`].
107    pub if_version_not_match: Option<String>,
108    /// Delete only when the file at the delete path still has the identity
109    /// recorded in this metadata.
110    ///
111    /// Pass metadata previously returned by OpenDAL for the same path.
112    /// OpenDAL derives a version match when the service supports version
113    /// conditions and the metadata contains a version. Otherwise it uses a
114    /// non-weak ETag if present. A changed or missing file fails the delete
115    /// with [`crate::ErrorKind::ConditionNotMatch`], as does combining this
116    /// option with a conflicting `if_match` or `if_version_match` value.
117    ///
118    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
119    /// service does not support the derived primitive condition, and
120    /// [`crate::ErrorKind::ConfigInvalid`] when the metadata contains neither
121    /// a version nor a non-weak ETag.
122    pub if_not_changed: Option<Metadata>,
123}

125/// Options for list operations.
126///
127/// # Groups
128/// - Traversal: `recursive`.
129/// - Pagination: `limit`, `start_after`.
130/// - Versioning: `versions`, `deleted`.
131///
132/// The operation fails with [`crate::ErrorKind::Unsupported`] when an option
133/// requires a capability that the operator lacks.
134#[derive(Debug, Clone, Default, Eq, PartialEq)]
135pub struct ListOptions {
136    /// Maximum results per request (backend hint) to control memory and throttling.
137    pub limit: Option<usize>,
138    /// Start listing after this key, excluding the key itself.
139    ///
140    /// Setting this requires [`Capability::list_with_start_after`](crate::Capability::list_with_start_after).
141    pub start_after: Option<String>,
142    /// Whether to list recursively under the prefix; default `false`.
143    pub recursive: bool,
144    /// Include object versions; default `false`.
145    ///
146    /// Requires [`Capability::list_with_versions`](crate::Capability::list_with_versions).
147    pub versions: bool,
148    /// Include delete markers; default `false`.
149    ///
150    /// Requires [`Capability::list_with_deleted`](crate::Capability::list_with_deleted).
151    pub deleted: bool,
152    /// Filter entries with a case-sensitive Unix shell style glob pattern.
153    ///
154    /// The pattern is relative to the listed directory, or to the parent
155    /// directory when the list path is a prefix without a trailing slash.
156    /// Entries must also satisfy the list prefix. The pattern determines which
157    /// depths can match, independently of [`Self::recursive`].
158    ///
159    /// - `*` matches zero or more characters within one path component.
160    /// - `?` matches one Unicode scalar value within one path component.
161    /// - `**` matches zero or more path components and must occupy a whole component.
162    /// - `{a,b}` matches either alternative within a component.
163    /// - `[ab]`, `[a-z]`, and `[!a-z]` match character sets, ranges, and negated sets.
164    /// - A backslash escapes the next character within a component.
165    /// - A trailing slash restricts matches to directories.
166    ///
167    /// Requires [`Capability::list_with_glob`](crate::Capability::list_with_glob). Standard operators use native
168    /// glob support when available and otherwise filter entries client-side.
169    /// Empty or malformed patterns fail with [`crate::ErrorKind::ConfigInvalid`].
170    pub glob: Option<String>,
171}

173/// Options for read operations.
174///
175/// Each condition checks the file being read, and every condition must hold
176/// for the read to proceed. A missing file fails the read with
177/// [`crate::ErrorKind::NotFound`] no matter which conditions are set. An
178/// existing file that fails a condition returns
179/// [`crate::ErrorKind::ConditionNotMatch`], and a condition the service does
180/// not advertise through its capability returns
181/// [`crate::ErrorKind::Unsupported`]; OpenDAL never silently drops a
182/// condition. A read with a zero limit or an empty range validates the read
183/// options and capabilities, then returns an empty buffer without accessing
184/// storage or evaluating conditions against the file.
185///
186/// See the [conditional operation specification][crate::docs::specs::conditional_operations]
187/// for the complete cross-operation contract.
188#[derive(Debug, Clone, Default, Eq, PartialEq)]
189pub struct ReadOptions {
190    /// Set `range` for this operation.
191    ///
192    /// If we have a file with size `n`.
193    ///
194    /// - `..` means read bytes in range `[0, n)` of file.
195    /// - `0..1024` and `..1024` means read bytes in range `[0, 1024)` of file
196    /// - `1024..` means read bytes in range `[1024, n)` of file
197    /// - `BytesRange::suffix(1024)` means read the last `min(1024, n)` bytes of file
198    ///
199    /// The type implements `From<RangeBounds<u64>>`, so users can use `(1024..).into()` instead.
200    pub range: BytesRange,
201    /// Return at most this many bytes from the start of the selected range,
202    /// accepting EOF after a valid starting byte.
203    ///
204    /// A non-zero limit on a forward selection starting at or beyond EOF
205    /// returns [`crate::ErrorKind::RangeNotSatisfied`]. A zero limit returns
206    /// an empty buffer after option and capability validation, without
207    /// accessing storage. Without a limit, bounded forward ranges require the
208    /// full requested length.
209    ///
210    /// Setting a limit with a positive `chunk`, with `concurrent` greater
211    /// than one, or when presigning a read returns [`crate::ErrorKind::ConfigInvalid`].
212    pub limit: Option<u64>,
213    /// Read the given version of the file instead of the current one.
214    ///
215    /// This selects which stored version the read returns; it is not a
216    /// condition on the file. A version that does not exist returns
217    /// [`crate::ErrorKind::NotFound`].
218    ///
219    /// Check [`crate::Capability::read_with_version`] before using this
220    /// option.
221    pub version: Option<String>,
222
223    /// Read only when the file has this exact ETag.
224    ///
225    /// The condition succeeds when the file exists and its ETag equals this
226    /// value. A file with a different ETag returns
227    /// [`crate::ErrorKind::ConditionNotMatch`]; a missing file returns
228    /// [`crate::ErrorKind::NotFound`]. Only concrete ETag values are
229    /// portable; a wildcard such as `"*"` has no portable meaning here.
230    ///
231    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
232    /// service does not advertise [`crate::Capability::read_with_if_match`].
233    pub if_match: Option<String>,
234    /// Read only when the file does not have this ETag.
235    ///
236    /// With a concrete ETag value, the condition succeeds when the file
237    /// exists with a different ETag. A file whose ETag equals this value
238    /// returns [`crate::ErrorKind::ConditionNotMatch`]; a missing file
239    /// returns [`crate::ErrorKind::NotFound`]. Only concrete ETag values
240    /// are portable; a wildcard such as `"*"` has no portable meaning here.
241    ///
242    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
243    /// service does not advertise
244    /// [`crate::Capability::read_with_if_none_match`].
245    pub if_none_match: Option<String>,
246    /// Read only when the file has this exact version.
247    ///
248    /// The condition succeeds when the file exists and its version equals
249    /// this value. A file with a different version returns
250    /// [`crate::ErrorKind::ConditionNotMatch`]; a missing file returns
251    /// [`crate::ErrorKind::NotFound`].
252    ///
253    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
254    /// service does not advertise
255    /// [`crate::Capability::read_with_if_version_match`].
256    pub if_version_match: Option<String>,
257    /// Read only when the file does not have this version.
258    ///
259    /// The condition succeeds when the file exists with a different version.
260    /// A file whose version equals this value returns
261    /// [`crate::ErrorKind::ConditionNotMatch`]; a missing file returns
262    /// [`crate::ErrorKind::NotFound`].
263    ///
264    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
265    /// service does not advertise
266    /// [`crate::Capability::read_with_if_version_not_match`].
267    pub if_version_not_match: Option<String>,
268    /// Read only when the file was modified after this timestamp.
269    ///
270    /// The condition succeeds when the file exists and was modified after
271    /// this time. A file not modified since this time returns
272    /// [`crate::ErrorKind::ConditionNotMatch`]; a missing file returns
273    /// [`crate::ErrorKind::NotFound`].
274    ///
275    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
276    /// service does not advertise
277    /// [`crate::Capability::read_with_if_modified_since`].
278    pub if_modified_since: Option<Timestamp>,
279    /// Read only when the file was not modified after this timestamp.
280    ///
281    /// The condition succeeds when the file exists and was not modified
282    /// after this time. A file modified after this time returns
283    /// [`crate::ErrorKind::ConditionNotMatch`]; a missing file returns
284    /// [`crate::ErrorKind::NotFound`].
285    ///
286    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
287    /// service does not advertise
288    /// [`crate::Capability::read_with_if_unmodified_since`].
289    pub if_unmodified_since: Option<Timestamp>,
290
291    /// Known content length of the object.
292    ///
293    /// This is an execution hint that allows OpenDAL to avoid extra metadata
294    /// requests while planning reads. It must not be used as an object identity
295    /// or consistency condition.
296    pub content_length_hint: Option<u64>,
297
298    /// Set `concurrent` for the operation.
299    ///
300    /// OpenDAL by default to read file without concurrent. This is not efficient for cases when users
301    /// read large chunks of data. By setting `concurrent`, opendal will reading files concurrently
302    /// on support storage services.
303    ///
304    /// By setting `concurrent`, opendal will fetch chunks concurrently with
305    /// the give chunk size.
306    ///
307    /// Refer to the [performance guide](https://github.com/apache/opendal/tree/main/core/core/src/docs/performance)
308    /// for more details.
309    pub concurrent: usize,
310    /// Set `chunk` for the operation.
311    ///
312    /// OpenDAL will use services' preferred chunk size by default. Users can set chunk based on their own needs.
313    /// `Some(0)` has the same behavior as `None`.
314    ///
315    /// Refer to the [performance guide](https://github.com/apache/opendal/tree/main/core/core/src/docs/performance)
316    /// for more details.
317    pub chunk: Option<usize>,
318    /// Controls the optimization strategy for range reads in [`crate::Reader::fetch`].
319    ///
320    /// When performing range reads, if the gap between two requested ranges is less than or
321    /// equal to the configured `gap` size, OpenDAL will merge these ranges into a single read request
322    /// and discard the unrequested data in between. This helps reduce the number of API calls
323    /// to remote storage services.
324    ///
325    /// Set to `0` to disable merging ranges separated by a gap. Overlapping or adjacent ranges
326    /// are still merged.
327    ///
328    /// This optimization is particularly useful when performing multiple small range reads
329    /// that are close to each other, as it reduces the overhead of multiple network requests
330    /// at the cost of transferring some additional data.
331    ///
332    /// Refer to the [performance guide](https://github.com/apache/opendal/tree/main/core/core/src/docs/performance)
333    /// for more details.
334    pub gap: Option<usize>,
335
336    /// Specify the content-type header that should be sent back by the operation.
337    ///
338    /// This option is only meaningful when used along with presign.
339    pub override_content_type: Option<String>,
340    /// Specify the `cache-control` header that should be sent back by the operation.
341    ///
342    /// This option is only meaningful when used along with presign.
343    pub override_cache_control: Option<String>,
344    /// Specify the `content-disposition` header that should be sent back by the operation.
345    ///
346    /// This option is only meaningful when used along with presign.
347    pub override_content_disposition: Option<String>,
348}

611/// Options for write operations.

617///   [`crate::ErrorKind::ConditionNotMatch`] and leaves the previously
618///   visible file unchanged. Depending on the service, the error may surface

884    /// ### Performance Impact

886    /// Setting appropriate chunk size can:
887    /// - Reduce number of API calls
888    /// - Improve overall throughput
